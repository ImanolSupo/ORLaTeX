\begin{ornotation}
  \set{I}{Items}
  \set{B}{Candidate bins}
  \param{w_i}{Item weight}\for{i in I}
  \param{C}{Bin capacity}
  \var{x_{ib}\in\{0,1\}}{Assignment}\for{i in I,b in B}
  \var{y_b\in\{0,1\}}{Bin used}\for{b in B}
\end{ornotation}
\begin{ormodel}
  \minimize \sum_{b\in B}y_b \label{bin-packing:0} \\
  \st \sum_{b\in B}x_{ib}=1 \for{i in I} \label{bin-packing:1} \\
  \sum_{i\in I}w_i x_{ib}<= C y_b \for{b in B} \label{bin-packing:2} \\
  x_{ib}\in\{0,1\} \for{i in I,b in B} \label{bin-packing:3} \\
  y_b\in\{0,1\} \for{b in B} \label{bin-packing:4}
\end{ormodel}

    % Delimited token replacement respects brace groups; never expand input.
    \tl_replace_all:Nnn #1 {>=}{\geq}

\cs_new_protected:Npn \__orlatex_domain_clause:
  {
    \seq_set_split:NnV \l__orlatex_domain_parts_seq {\in}\l__orlatex_domain_clause_tl
    \int_compare:nNnTF {\seq_count:N\l__orlatex_domain_parts_seq}={2}
      {\seq_map_inline:Nn \l__orlatex_domain_parts_seq
        {\tl_if_blank:nT {##1}

    \tl_clear:N \l__orlatex_domain_clause_tl
  }
\cs_new_protected:Npn \__orlatex_domain_scan:
  {
    \tl_trim_spaces:N \l__orlatex_domain_scan_tl
    \tl_if_empty:NF \l__orlatex_domain_scan_tl
      {
        \tl_if_head_is_group:VTF \l__orlatex_domain_scan_tl
          {\tl_put_right:Nx \l__orlatex_domain_clause_tl
            {{\tl_head:N\l__orlatex_domain_scan_tl}}}
          {
            \tl_if_head_eq_meaning:VNT \l__orlatex_domain_scan_tl (
              {\int_incr:N\l__orlatex_domain_parentheses_int}
            \tl_if_head_eq_meaning:VNT \l__orlatex_domain_scan_tl )
              {\int_decr:N\l__orlatex_domain_parentheses_int}
            \tl_if_head_eq_meaning:VNTF \l__orlatex_domain_scan_tl ,
              {\int_compare:nNnTF {\l__orlatex_domain_parentheses_int}={0}
                {\__orlatex_domain_clause:}
                {\tl_put_right:Nn\l__orlatex_domain_clause_tl {,}}}
              {\tl_put_right:Nx\l__orlatex_domain_clause_tl
                {\tl_head:N\l__orlatex_domain_scan_tl}}
          }
        \tl_set:Nx \l__orlatex_domain_scan_tl {\tl_tail:N\l__orlatex_domain_scan_tl}
        \__orlatex_domain_scan:
      }
  }
\cs_new_protected:Npn \__orlatex_input_domain:nN #1#2
  {
    \tl_set:Nn #2 {#1}
    \tl_trim_spaces:N #2
    \bool_if:NF \l__orlatex_input_raw_bool
      {
        \tl_replace_all:Nnn #2 {~in~}{\in}
        \__orlatex_input_normalize:N #2
        \tl_set_eq:NN \l__orlatex_domain_scan_tl #2
        \tl_clear:N\l__orlatex_domain_clause_tl
        \int_zero:N\l__orlatex_domain_parentheses_int
        \__orlatex_domain_scan:
        \__orlatex_domain_clause:
        \int_compare:nNnF {\l__orlatex_domain_parentheses_int}={0}
          {\msg_error:nnn {orlatex}{input}{unbalanced~domain~tuple;~use~\string\for*~for~raw~math}}
      }
  }

\cs_new_protected:Npn \__orlatex_input_scan:
  {
    \tl_trim_spaces:N \l__orlatex_input_tl
    \tl_if_empty:NF \l__orlatex_input_tl
      {
        \tl_if_head_eq_meaning:VNT \l__orlatex_input_tl \objective
          {\msg_error:nnn {orlatex}{input}{do~not~mix~structured~and~lightweight~rows}}
        \tl_if_head_eq_meaning:VNT \l__orlatex_input_tl \constraint
          {\msg_error:nnn {orlatex}{input}{do~not~mix~structured~and~lightweight~rows}}
        \tl_if_head_is_group:VTF \l__orlatex_input_tl
          {\tl_put_right:Nx \l__orlatex_input_expression_tl
             {{\tl_head:N \l__orlatex_input_tl}}
           \__orlatex_input_drop:}
          {
            \tl_if_head_eq_meaning:VNTF \l__orlatex_input_tl \for
              {\__orlatex_input_drop:
               \bool_set_false:N \l__orlatex_input_raw_bool
               \tl_if_head_eq_meaning:VNT \l__orlatex_input_tl *
                 {\bool_set_true:N \l__orlatex_input_raw_bool\__orlatex_input_drop:}
               \tl_if_empty:NF \l__orlatex_input_domain_tl
                 {\msg_error:nnn {orlatex}{input}{duplicate~domain}}
               \__orlatex_input_argument:N \l_tmpa_tl
               \exp_args:NV \__orlatex_input_domain:nN \l_tmpa_tl
                 \l__orlatex_input_domain_tl}
              {\tl_if_head_eq_meaning:VNTF \l__orlatex_input_tl \label
                {\__orlatex_input_drop:
                 \tl_if_empty:NF \l__orlatex_input_label_tl
                   {\msg_error:nnn {orlatex}{input}{duplicate~label}}
                 \__orlatex_input_argument:N \l__orlatex_input_label_tl}
                {\tl_if_head_eq_meaning:VNTF \l__orlatex_input_tl \notag
                  {\bool_set_true:N \l__orlatex_input_notag_bool\__orlatex_input_drop:}
                  {\tl_put_right:Nx \l__orlatex_input_expression_tl
                     {\tl_head:N \l__orlatex_input_tl}
                   \__orlatex_input_drop:}}}
          }
        \__orlatex_input_scan:
      }
  }
